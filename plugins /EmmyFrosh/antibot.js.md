const fs = require("fs");
const path = require("path");

// ─────────────────────────────────────────────────────────────────────────────
// ANTIBOT
// Detects other bots answering commands in a group, deletes their reply, and
// optionally warns / kicks them.
//
//   .antibot status
//   .antibot off | delete | warn | kick
//   .antibot limit <n>          (warnings before kick, warn mode)
//   .antibot allow <number>     (never touch this account — e.g. a bot you trust)
//   .antibot unallow <number>
//   .antibot resetwarns
// ─────────────────────────────────────────────────────────────────────────────

const DB_FILE = path.join(process.cwd(), "antibot-data.json");

const CMD_TTL = 120_000; // how long a message stays "known" so a quote of it is recognised
const UNQUOTED_WINDOW = 20_000; // other-account bot replies must land this soon after chat activity
const PUNISH_DEBOUNCE = 8_000; // a burst of replies from one bot counts as ONE offence
const DEFAULT_WARN_LIMIT = 3;

// Self-bots reply from the SAME account that typed the command. They often quote
// a FAKE message (e.g. a "Meta AI" card), so the quote can't be trusted — we go by
// who sent it and how soon after the account's own message it arrived.
const SAME_ACCOUNT_WINDOW = 45_000;
const QUICK_HEAVY_WINDOW = 4_000; // a photo/menu/long reply this soon after the message it quotes = bot
const QUICK_WEB_WINDOW = 3_000; // a web-style id replying this soon after the message it quotes = bot
const SAME_ACCOUNT_QUICK = 10_000; // quoting its OWN message: must answer within this
const SHORT_COMMAND_MAX = 80; // …and that message must be short (a command, not a chat line)
const SEEN_TTL = 60 * 60_000; // remember message ids for an hour (so quotes of them are "real")
const MEMBERS_TTL = 120_000; // refresh the group member list in the background this often

let debug = false;
const stats = { seen: 0, lastAt: 0 };

// Any punctuation/symbol prefix (. ! # / $ & * + ? ~ - …) directly followed by a
// word. "@" is excluded so plain mentions aren't treated as commands.
const PREFIX_RE = /^(?!@)[\p{P}\p{S}]{1,3}[\p{L}\p{N}]/u;

// Message IDs generated by Baileys-based bots (humans on official clients use
// different formats). Best-effort heuristic — see notes in the reply.
function looksLikeBotId(id = "") {
  return (
    /^BAE5[0-9A-F]{12}$/i.test(id) || // older Baileys (16 chars)
    /^3EB0[0-9A-F]{8}$/i.test(id) || // older Baileys (12 chars)
    /^3EB0[0-9A-F]{36}$/i.test(id) // current Baileys (40 chars)
  );
}

// Looser "3EB0…" style id (Baileys' newer generator mimics WhatsApp Web's format, so
// on its own this proves nothing — it's only used together with other evidence).
function looksWebStyleId(id = "") {
  return /^3EB0[0-9A-F]{16,24}$/i.test(id);
}

// ── Storage ──────────────────────────────────────────────────────────────────
function loadDb() {
  try {
    const d = JSON.parse(fs.readFileSync(DB_FILE, "utf8"));
    if (d && typeof d.groups === "object") return d;
  } catch {}
  return { groups: {} };
}
let db = loadDb();

function saveDb() {
  try {
    fs.writeFileSync(DB_FILE, JSON.stringify(db, null, 2));
  } catch (e) {
    console.error("antibot: could not save settings:", e.message);
  }
}

function groupCfg(chat) {
  if (!db.groups[chat]) {
    db.groups[chat] = { mode: "off", warnLimit: DEFAULT_WARN_LIMIT, allow: [], warns: {} };
  }
  return db.groups[chat];
}

// ── JID helpers ──────────────────────────────────────────────────────────────
const norm = (jid = "") => String(jid || "").replace(/:\d+(?=@)/, "");
const num = (jid = "") => norm(jid).split("@")[0];

function ownKeys(ednut) {
  return [ednut.user?.id, ednut.user?.lid].filter(Boolean).map(norm);
}
function isOwnAccount(ednut, keys) {
  const mine = ownKeys(ednut);
  return keys.some((k) => mine.includes(k));
}
function senderKeysOf(m) {
  return [norm(m.key?.participant || m.participant), norm(m.key?.participantAlt)].filter(Boolean);
}
function isAdminIn(meta, keys) {
  const p = (meta?.participants || []).find((p) =>
    [p.id, p.lid, p.phoneNumber, p.jid].filter(Boolean).map(norm).some((j) => keys.includes(j))
  );
  return !!p?.admin;
}

// ── Message parsing ──────────────────────────────────────────────────────────
const IGNORED_TYPES = new Set([
  "protocolMessage",
  "reactionMessage",
  "encReactionMessage",
  "pollUpdateMessage",
  "editedMessage",
  "keepInChatMessage",
]);

function unwrap(message) {
  let msg = message;
  for (let i = 0; i < 5 && msg; i++) {
    const inner =
      msg.ephemeralMessage?.message ||
      msg.viewOnceMessage?.message ||
      msg.viewOnceMessageV2?.message ||
      msg.documentWithCaptionMessage?.message;
    if (!inner) break;
    msg = inner;
  }
  return msg;
}

function getContent(message) {
  const msg = unwrap(message);
  if (!msg) return null;
  const type = Object.keys(msg).find(
    (k) => k !== "messageContextInfo" && k !== "senderKeyDistributionMessage"
  );
  if (!type) return null;
  return { type, node: msg[type] };
}

function textOf({ node }) {
  if (typeof node === "string") return node;
  return node?.text || node?.caption || "";
}

// ── Runtime state ────────────────────────────────────────────────────────────
const cmds = new Map(); // `${chat}|${id}` -> { keys, ts, len }   (lookup by quoted id)
const humans = new Map(); // chat -> [{ id, keys, ts, len }]       (recent human messages)
const lastPunished = new Map(); // `${chat}|${jid}` -> ts
const history = []; // last few decisions, shown by ".antibot last"
const seen = new Map(); // `${chat}|${id}` -> { ts, keys, text }  (EVERY message we've seen)
const members = new Map(); // chat -> { ids:Set, nums:Set, ts }
const inflight = new Set();
let lastPrune = 0;

function cacheMembers(chat, meta) {
  const ids = new Set();
  const nums = new Set();
  for (const p of meta?.participants || []) {
    for (const j of [p.id, p.lid, p.phoneNumber, p.jid]) {
      if (!j) continue;
      ids.add(norm(j));
      nums.add(num(j));
    }
  }
  if (ids.size) members.set(chat, { ids, nums, ts: Date.now() });
}

// Refresh in the background — classification itself never waits for this.
function refreshMembers(ednut, chat, now) {
  const c = members.get(chat);
  if ((c && now - c.ts < MEMBERS_TTL) || inflight.has(chat) || !ednut.groupMetadata) return;
  inflight.add(chat);
  Promise.resolve(ednut.groupMetadata(chat))
    .then((meta) => cacheMembers(chat, meta))
    .catch(() => {})
    .finally(() => inflight.delete(chat));
}

// A quote whose author is NOT in the group is a spoofed quote (e.g. a fake
// "Meta AI" card). If we can't verify membership we say "no" — never guess.
function isForeignParticipant(chat, participant) {
  const c = members.get(chat);
  if (!c || !participant) return false;
  const p = norm(participant);
  return !c.ids.has(p) && !c.nums.has(num(p));
}

// ── Quote verification ───────────────────────────────────────────────────────
function quotedTextOf(qm) {
  if (!qm || typeof qm !== "object") return "";
  const first = Object.values(qm).find((v) => v && typeof v === "object");
  return (
    (typeof qm.conversation === "string" && qm.conversation) ||
    qm.extendedTextMessage?.text ||
    first?.caption ||
    first?.text ||
    ""
  );
}
const squash = (t = "") => String(t).toLowerCase().replace(/\s+/g, " ").trim();
const domainOf = (j = "") => String(j).split("@")[1] || "";

// Returns "none" | "real" | "fake" | "unseen" — and WHY, so it shows in .antibot last.
//   real   : points at a message we saw, sent by who the quote says, with the same text
//   fake   : wrong sender, different text, or an unknown message "from" someone not in the group
//   unseen : an older message we never saw (a person replying to old chat) — never treated as fake
function verifyQuote(chat, ctx) {
  const id = ctx?.stanzaId;
  if (!id) return { state: "none", note: "" };
  const rec = seen.get(`${chat}|${id}`);

  if (!rec) {
    return isForeignParticipant(chat, ctx.participant)
      ? { state: "fake", note: "quotes a message from someone who isn't in this group" }
      : { state: "unseen", note: "" };
  }

  // 1) Sender check — only when both sides use the same JID type (pn vs lid can't be compared).
  const qp = ctx.participant ? norm(ctx.participant) : "";
  if (qp && rec.keys.length) {
    const comparable = rec.keys.filter((k) => domainOf(k) === domainOf(qp));
    if (comparable.length && !comparable.includes(qp)) {
      return { state: "fake", note: "quote credits the message to the wrong sender" };
    }
  }
  // 2) Text check — a genuine quote carries the original text (edits/truncation tolerated).
  const qt = squash(quotedTextOf(ctx.quotedMessage));
  const ot = squash(rec.text);
  if (qt && ot && !(qt === ot || qt.includes(ot) || ot.includes(qt))) {
    return { state: "fake", note: "quoted text doesn't match the real message" };
  }
  return { state: "real", note: "" };
}

// Things a bot sends and a person quick-replying to themselves basically never does.
const BOTLIKE_TYPES = new Set([
  "imageMessage", "videoMessage", "audioMessage", "stickerMessage", "documentMessage",
  "locationMessage", "contactMessage", "contactsArrayMessage", "productMessage",
  "buttonsMessage", "listMessage", "templateMessage", "interactiveMessage",
]);
// "Heavy" replies: media a person cannot produce within a few seconds of a message.
// (Stickers and voice notes are excluded — people really do send those instantly.)
const HEAVY_TYPES = new Set([
  "imageMessage", "videoMessage", "documentMessage", "locationMessage", "contactMessage",
  "contactsArrayMessage", "productMessage", "buttonsMessage", "listMessage",
  "templateMessage", "interactiveMessage",
]);
function heavyContent(content) {
  // Media/buttons only. Text length is never treated as bot evidence — people write long messages.
  return HEAVY_TYPES.has(content.type);
}

function botLikeContent(content) {
  if (BOTLIKE_TYPES.has(content.type)) return true;
  const n = content.node;
  if (n && typeof n === "object" && n.contextInfo?.isForwarded) return true;
  return false; // text length is never evidence
}
const hooked = new WeakSet();

function prune(now) {
  if (now - lastPrune < 1000) return;
  lastPrune = now;
  for (const [k, v] of seen) {
    if (now - v.ts > SEEN_TTL) seen.delete(k);
    else break; // insertion order ≈ time order
  }
  for (const [k, v] of cmds) if (now - v.ts > CMD_TTL) cmds.delete(k);
  for (const [chat, list] of humans) {
    const kept = list.filter((c) => now - c.ts <= CMD_TTL);
    if (kept.length) humans.set(chat, kept);
    else humans.delete(chat);
  }
  for (const [k, ts] of lastPunished) if (now - ts > 60_000) lastPunished.delete(k);
}

// Every human message is remembered, whatever prefix (or none) it uses.
function remember(chat, id, keys, now, text) {
  cmds.set(`${chat}|${id}`, { keys, ts: now, len: text.length });
  const list = humans.get(chat) || [];
  list.push({ id, keys, ts: now, len: text.length });
  humans.set(chat, list.slice(-30));
}

function note(entry) {
  history.push(entry);
  if (history.length > 10) history.shift();
}

// ── Actions ──────────────────────────────────────────────────────────────────
async function deleteMsg(ednut, chat, key) {
  for (let attempt = 1; attempt <= 2; attempt++) {
    try {
      await ednut.sendMessage(chat, { delete: key });
      return true;
    } catch (e) {
      if (attempt === 2) {
        console.error("antibot: delete failed (is the bot an admin?):", e.message || e);
        return false;
      }
      await new Promise((r) => setTimeout(r, 250));
    }
  }
}

async function kickUser(ednut, chat, jid) {
  try {
    await ednut.groupParticipantsUpdate(chat, [jid], "remove");
    return true;
  } catch (e) {
    console.error("antibot: kick failed (is the bot an admin?):", e.message || e);
    return false;
  }
}

async function punish(ednut, chat, cfg, jid, reason, now) {
  if (cfg.mode === "delete") return;

  // A burst of replies from the same bot counts as a single offence.
  const dKey = `${chat}|${norm(jid)}`;
  const last = lastPunished.get(dKey);
  if (last && now - last < PUNISH_DEBOUNCE) return;
  lastPunished.set(dKey, now);

  const tag = `@${num(jid)}`;
  const mentions = [jid];

  if (cfg.mode === "kick") {
    const ok = await kickUser(ednut, chat, jid);
    await ednut.sendMessage(chat, {
      text: ok
        ? `🚫 ${tag} was removed — ${reason}.`
        : `⚠️ ${tag} — ${reason}, but I couldn't remove them. Make me an admin.`,
      mentions,
    });
    return;
  }

  if (cfg.mode === "warn") {
    const id = num(jid);
    cfg.warns[id] = (cfg.warns[id] || 0) + 1;
    const count = cfg.warns[id];
    const limit = cfg.warnLimit || DEFAULT_WARN_LIMIT;

    if (count >= limit) {
      delete cfg.warns[id];
      saveDb();
      const ok = await kickUser(ednut, chat, jid);
      await ednut.sendMessage(chat, {
        text: ok
          ? `🚫 ${tag} reached *${limit}/${limit}* warnings and was removed.`
          : `⚠️ ${tag} reached *${limit}/${limit}* warnings, but I couldn't remove them. Make me an admin.`,
        mentions,
      });
    } else {
      saveDb();
      await ednut.sendMessage(chat, {
        text: `⚠️ ${tag} — ${reason}.\nWarning *${count}/${limit}*`,
        mentions,
      });
    }
  }
}

// ── Core detection (synchronous — no network calls in here, so it never queues) ─
function classify(ednut, m) {
  const chat = m.key?.remoteJid;
  if (!chat || !chat.endsWith("@g.us")) return;

  const content = getContent(m.message);
  if (!content || IGNORED_TYPES.has(content.type)) return;

  const now = Date.now();
  prune(now);

  // Remember EVERY message — id, sender and text — so any later quote of it can be verified.
  seen.set(`${chat}|${m.key.id}`, {
    ts: now,
    keys: m.key.fromMe ? [] : senderKeysOf(m),
    text: (textOf(content) || "").slice(0, 500),
  });

  // Never trigger on the host bot's own messages.
  if (m.key.fromMe) return;

  const cfg = db.groups[chat];
  if (!cfg || cfg.mode === "off") return;

  const keys = senderKeysOf(m);
  if (keys.length === 0) return;
  if (isOwnAccount(ednut, keys)) return; // …or from the host bot's own account

  const allow = cfg.allow || [];
  if (keys.some((k) => allow.includes(num(k)))) return;

  refreshMembers(ednut, chat, now);

  const ctx = typeof content.node === "object" ? content.node?.contextInfo : null;
  const quotedId = ctx?.stanzaId || null;
  const botId = looksLikeBotId(m.key.id);
  const known = quotedId ? cmds.get(`${chat}|${quotedId}`) : null; // a recent human text message
  // ALWAYS verify the quote before judging the message.
  const vq = verifyQuote(chat, ctx);
  const quoteState = vq.state;
  const quoteIsReal = quoteState === "real";
  const fakeQuote = quoteState === "fake";

  const list = humans.get(chat) || [];
  const latest = (pred) => [...list].reverse().find((c) => c.id !== m.key.id && pred(c));
  const ownRecent = latest((c) => keys.some((k) => c.keys.includes(k)) && now - c.ts <= SAME_ACCOUNT_WINDOW);
  const anyRecent = latest((c) => now - c.ts <= UNQUOTED_WINDOW);
  const sameAsKnown = !!known && keys.some((k) => known.keys.includes(k));
  const forwarded = !!(ctx?.isForwarded || ctx?.forwardingScore > 0);
  const sinceKnown = known ? now - known.ts : Infinity;
  const sameAccount = sameAsKnown || !!ownRecent;

  // A message is only treated as a bot reply on POSITIVE evidence of a bot.
  // Merely replying to someone — however fast — is never enough.
  let isBotReply = false;
  let why = "looks like a person";

  if (botId && known) {
    isBotReply = true;
    why = "bot-style id + quotes a command";
  } else if (botId && fakeQuote) {
    isBotReply = true;
    why = "bot-style id + spoofed quote";
  } else if (botId && !quotedId && anyRecent) {
    isBotReply = true;
    why = "bot-style id right after chat";
  } else if (forwarded && quoteIsReal) {
    // A reply that is ALSO marked "Forwarded" doesn't come from a normal client.
    isBotReply = true;
    why = "reply marked Forwarded (bots do this to look official)";
  } else if (!sameAsKnown && known && heavyContent(content) && sinceKnown <= QUICK_HEAVY_WINDOW) {
    isBotReply = true;
    why = `photo/menu/long reply only ${Math.round(sinceKnown)}ms after the message it quotes`;
  } else if (!sameAsKnown && known && looksWebStyleId(m.key.id) && sinceKnown <= QUICK_WEB_WINDOW) {
    isBotReply = true;
    why = `web-style id replying ${Math.round(sinceKnown)}ms after the message it quotes`;
  } else if (fakeQuote && anyRecent) {
    // A normal client can't forge a quote (wrong sender / wrong text / author not in the group).
    isBotReply = true;
    why = `forged quote (${vq.note})`;
  } else if (!botId && fakeQuote && ownRecent) {
    isBotReply = true;
    why = "spoofed quote from the account that just sent a command";
  } else if (
    !botId &&
    sameAsKnown &&
    now - known.ts <= SAME_ACCOUNT_QUICK &&
    known.len <= SHORT_COMMAND_MAX &&
    botLikeContent(content)
  ) {
    isBotReply = true;
    why = "same account answered its own short message with media/forwarded content";
  }

  note({
    id: m.key.id,
    from: num(keys[0]),
    quoted: quoteState,
    botId,
    sameAccount,
    verdict: isBotReply ? "DELETE" : "ignore",
    why,
    type: content.type,
    forwarded,
    ms: Number.isFinite(sinceKnown) ? Math.round(sinceKnown) : null,
    fakeWhy: vq.note,
  });
  if (debug) {
    console.log(
      `[antibot] id=${m.key.id} from=${num(keys[0])} quoted=${quoteState} ` +
        `botId=${botId} sameAccount=${sameAccount} -> ${isBotReply ? "BOT REPLY (" + why + ")" : "ignored"}`
    );
  }

  if (!isBotReply) {
    // A human message: remember it so later replies can be matched against it.
    if (!botId) {
      const text = textOf(content).trim();
      if (text) remember(chat, m.key.id, keys, now, text);
    }
    return;
  }

  return {
    chat,
    cfg,
    key: m.key,
    jid: m.key.participant || m.participant,
    reason: sameAccount ? "self-bot replies aren't allowed here" : "bots aren't allowed here",
    now,
  };
}

// Delete goes out FIRST and immediately; the warn/kick runs alongside it in the
// background so it can never hold up the delete (or the next message).
function act(ednut, a) {
  deleteMsg(ednut, a.chat, a.key).catch(() => {});
  punish(ednut, a.chat, a.cfg, a.jid, a.reason, a.now).catch((e) =>
    console.error("antibot punish error:", e)
  );
}

function onUpsert(ednut, { messages, type }) {
  if (type !== "notify") return;
  stats.seen += (messages || []).length;
  stats.lastAt = Date.now();
  for (const m of messages || []) {
    try {
      const action = classify(ednut, m);
      if (action) act(ednut, action); // not awaited
    } catch (e) {
      console.error("antibot error:", e);
    }
  }
}

function ensureListener(ednut) {
  if (!ednut?.ev?.on) return false;
  if (hooked.has(ednut)) return true;
  hooked.add(ednut);
  ednut.ev.on("messages.upsert", (u) => onUpsert(ednut, u));
  return true;
}

// If the bot exposes its socket globally, hook in as soon as the plugin loads.
try {
  ensureListener(global.ednut || global.conn || global.sock);
} catch {}

// ── Command ──────────────────────────────────────────────────────────────────
function statusText(cfg, botAdmin) {
  const modeLabel = {
    off: "🔴 Off",
    delete: "🗑️ Delete only",
    warn: "⚠️ Delete + warn",
    kick: "🚫 Delete + kick",
  }[cfg.mode];
  let t = `🤖 *ANTIBOT*\n\n`;
  t += `📌 Mode: *${modeLabel}*\n`;
  t += `⚠️ Warn limit: *${cfg.warnLimit}*\n`;
  t += `✅ Allowed accounts: ${cfg.allow.length ? cfg.allow.map((n) => `+${n}`).join(", ") : "none"}\n`;
  t += `🛡️ Bot admin: ${botAdmin ? "yes" : "*no* — I can't delete or kick until I'm admin"}`;
  return t;
}

const HELP =
  `🤖 *ANTIBOT*\n\n` +
  `• .antibot status\n` +
  `• .antibot off | delete | warn | kick\n` +
  `• .antibot limit <n> — warnings before kick\n` +
  `• .antibot allow <number> — never touch this account\n` +
  `• .antibot unallow <number>\n` +
  `• .antibot resetwarns\n` +
  `• .antibot debug — toggle console logging + show listener status\n` +
  `• .antibot last — show how the last messages were judged`;

module.exports = [
  {
    command: ["antibot"],
    alias: ["ab"],
    description: "Delete/warn/kick other bots that respond to commands",
    category: "Group",
    filename: __filename,
    async execute(m, { ednut, q, reply, from }) {
      ensureListener(ednut);

      if (!from.endsWith("@g.us")) return reply("❌ This command only works in groups.");

      const args = (q || "").trim().split(/\s+/).filter(Boolean);
      const sub = (args[0] || "status").toLowerCase();
      const cfg = groupCfg(from);

      let meta;
      try {
        meta = await ednut.groupMetadata(from);
      } catch (e) {
        return reply(`❌ Couldn't read group info.\n\n${e.message || e}`);
      }
      cacheMembers(from, meta);
      const botAdmin = isAdminIn(meta, ownKeys(ednut));

      if (sub === "status") return reply(statusText(cfg, botAdmin));

      if (!m.key.fromMe && !isAdminIn(meta, senderKeysOf(m))) {
        return reply("❌ Only group admins can change antibot settings.");
      }

      if (["off", "delete", "warn", "kick"].includes(sub)) {
        cfg.mode = sub;
        saveDb();
        let t = statusText(cfg, botAdmin);
        if (sub !== "off" && !botAdmin) {
          t += `\n\n⚠️ Make me a group admin or I can't delete messages or remove anyone.`;
        }
        return reply(t);
      }

      if (sub === "limit") {
        const n = parseInt(args[1], 10);
        if (!Number.isInteger(n) || n < 1 || n > 20) {
          return reply("❌ Give a number between 1 and 20.\nExample: .antibot limit 3");
        }
        cfg.warnLimit = n;
        saveDb();
        return reply(`✅ Warn limit set to *${n}*. A bot is kicked on warning ${n}.`);
      }

      if (sub === "allow" || sub === "unallow") {
        const content = getContent(m.message);
        const mentioned =
          typeof content?.node === "object" ? content.node?.contextInfo?.mentionedJid?.[0] : null;
        const number = mentioned ? num(mentioned) : (args[1] || "").replace(/\D/g, "");
        if (!number) {
          return reply(`❌ Give a number or @mention.\nExample: .antibot ${sub} 2348012345678`);
        }
        if (sub === "allow") {
          if (!cfg.allow.includes(number)) cfg.allow.push(number);
          saveDb();
          return reply(`✅ +${number} is now exempt from antibot.`);
        }
        cfg.allow = cfg.allow.filter((n) => n !== number);
        saveDb();
        return reply(`✅ +${number} removed from the exempt list.`);
      }

      if (sub === "debug") {
        debug = !debug;
        const attached = ensureListener(ednut);
        const ago = stats.lastAt ? `${Math.round((Date.now() - stats.lastAt) / 1000)}s ago` : "never";
        return reply(
          `🛠️ *ANTIBOT DEBUG* — logging ${debug ? "ON" : "OFF"}\n\n` +
            `• Listener attached: ${attached ? "yes" : "*NO* (ednut.ev not found)"}\n` +
            `• Messages seen since start: ${stats.seen}\n` +
            `• Last message seen: ${ago}\n\n` +
            (debug
              ? `Now trigger the other bot and check the console for [antibot] lines.`
              : `Console logging stopped.`)
        );
      }

      if (sub === "last") {
        if (history.length === 0) return reply("📭 Nothing inspected yet. Send a command and try again.");
        const lines = history
          .slice(-8)
          .map(
            (h, i) =>
              `${i + 1}. ${h.verdict === "DELETE" ? "🗑️" : "▫️"} +${h.from}\n` +
              `   id: ${h.id}\n` +
              `   type: ${h.type} • botId: ${h.botId} • quote: ${h.quoted} • fwd: ${h.forwarded}` +
              `${h.ms !== null ? ` • ${h.ms}ms after quoted msg` : ""}\n` +
              `${h.fakeWhy ? `   fake quote: ${h.fakeWhy}\n` : ""}` +
              `   → ${h.why}`
          )
          .join("\n");
        return reply(`🔎 *LAST INSPECTED MESSAGES*\n\n${lines}`);
      }

      if (sub === "resetwarns") {
        cfg.warns = {};
        saveDb();
        return reply("✅ All antibot warnings in this group were cleared.");
      }

      return reply(HELP);
    },
  },
